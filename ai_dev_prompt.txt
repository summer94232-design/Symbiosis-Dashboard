interface CacheEntry {
  data: any
  timestamp: number
  ttl: number
}

const store = new Map<string, CacheEntry>()

export const cache = {
  get<T>(key: string): T | null {
    const entry = store.get(key)
    if (!entry) return null
    if (Date.now() - entry.timestamp > entry.ttl) {
      store.delete(key)
      return null
    }
    return entry.data as T
  },

  set(key: string, data: any, ttlSeconds = 60) {
    store.set(key, { data, timestamp: Date.now(), ttl: ttlSeconds * 1000 })
  },

  invalidate(key: string) {
    store.delete(key)
  },

  invalidatePrefix(prefix: string) {
    store.forEach((_, k) => { if (k.startsWith(prefix)) store.delete(k) })
  }
}

export const FEE_RATE = 0.001425
export const FEE_DISCOUNT = 0.6  // 6折
export const MIN_FEE = 20
export const TAX_RATE_STOCK = 0.003
export const TAX_RATE_ETF = 0.001

export const isETF = (code: string) => code.replace('.TW', '').startsWith('00')

// 計算手續費（6折，無條件捨去，最低20元）
export function calcFee(price: number, shares: number): number {
  const amount = price * shares
  const fee = Math.floor(amount * FEE_RATE * FEE_DISCOUNT)
  return Math.max(fee, MIN_FEE)
}

// 計算交易稅（賣出才有）
export function calcTax(price: number, shares: number, code: string): number {
  const amount = price * shares
  const rate = isETF(code) ? TAX_RATE_ETF : TAX_RATE_STOCK
  return Math.floor(amount * rate)
}

export interface TradeRecord {
  date: string
  type: 'buy' | 'sell'
  code: string
  name: string
  price: number
  shares: number
  fee: number
  tax: number
  total: number
  note?: string
}

export interface HoldingPosition {
  code: string
  name: string
  shares: number
  avgCost: number
  totalCost: number
}

export interface RealizedRecord {
  date: string
  code: string
  name: string
  shares: number
  avgCost: number
  sellPrice: number
  realized: number
  returnRate: number
}

export interface CalcResult {
  holdings: Record<string, HoldingPosition>
  realized: RealizedRecord[]
}

export function calcPositions(trades: TradeRecord[]): CalcResult {
  const sorted = [...trades].sort((a, b) => a.date.localeCompare(b.date))
  const holdings: Record<string, HoldingPosition> = {}
  const realized: RealizedRecord[] = []

  for (const t of sorted) {
    const { code, name, type, price, shares, fee, tax } = t
    if (!holdings[code]) {
      holdings[code] = { code, name, shares: 0, avgCost: 0, totalCost: 0 }
    }
    const pos = holdings[code]

    if (type === 'buy') {
      const buyCost = price * shares + fee
      const newTotalCost = pos.totalCost + buyCost
      const newShares = pos.shares + shares
      pos.avgCost = newShares > 0 ? newTotalCost / newShares : 0
      pos.totalCost = newTotalCost
      pos.shares = newShares
      pos.name = name
    } else {
      const sellRevenue = price * shares - fee - tax
      const costBasis = pos.avgCost * shares
      const pnl = sellRevenue - costBasis
      const returnRate = costBasis > 0 ? (pnl / costBasis) * 100 : 0
      realized.push({
        date: t.date, code, name, shares,
        avgCost: pos.avgCost, sellPrice: price,
        realized: pnl, returnRate
      })
      pos.shares -= shares
      pos.totalCost = pos.avgCost * pos.shares
      if (pos.shares <= 0) {
        pos.shares = 0
        pos.totalCost = 0
        pos.avgCost = 0
      }
    }
  }

  const activeHoldings: Record<string, HoldingPosition> = {}
  Object.entries(holdings).forEach(([code, pos]) => {
    if (pos.shares > 0) activeHoldings[code] = pos
  })

  return { holdings: activeHoldings, realized: realized.reverse() }
}

export function calcAvailableCash(
  trades: TradeRecord[],
  cashRecords: { type: 'in' | 'out'; amount: number }[]
): number {
  const netCash = cashRecords.reduce((s, r) => r.type === 'in' ? s + r.amount : s - r.amount, 0)
  const buyTotal = trades.filter(t => t.type === 'buy').reduce((s, t) => s + t.total, 0)
  const sellTotal = trades.filter(t => t.type === 'sell').reduce((s, t) => s + t.total, 0)
  return netCash - buyTotal + sellTotal
}

// 計算台灣期貨結算日（每月第三個週三）
export function getFuturesSettlementDates(year: number, month: number): CalendarEvent[] {
  const results: CalendarEvent[] = []
  for (let m = 1; m <= 12; m++) {
    if (year === new Date().getFullYear() - 1) continue
    let count = 0
    for (let d = 1; d <= 31; d++) {
      const date = new Date(year, m - 1, d)
      if (date.getMonth() !== m - 1) break
      if (date.getDay() === 3) { // 週三
        count++
        if (count === 3) {
          const dateStr = `${year}-${String(m).padStart(2, '0')}-${String(d).padStart(2, '0')}`
          results.push({
            id: `futures-${year}-${m}`,
            date: dateStr,
            title: '台指期結算日',
            type: 'futures',
            description: '台灣加權指數期貨結算'
          })
          break
        }
      }
    }
  }
  return results
}

const FUGLE_API_KEY = import.meta.env.VITE_FUGLE_API_KEY

// 取得單一股票報價
export async function getQuote(symbol: string) {
  try {
    const res = await fetch(
      `https://api.fugle.tw/marketdata/v1.0/stock/intraday/quote/${symbol}`,
      { headers: { 'X-API-KEY': FUGLE_API_KEY } }
    )
    if (!res.ok) return null
    const data = await res.json()
    return data
  } catch {
    return null
  }
}

// 從報價資料取得最佳現價
// 優先順序：closePrice → lastPrice → referencePrice
export function extractPrice(quote: any): number | null {
  if (!quote) return null
  return quote.closePrice || quote.lastPrice || quote.referencePrice || null
}

// 批次取得多檔股票現價
export async function getMultipleQuotes(symbols: string[]): Promise<Record<string, number | null>> {
  const results: Record<string, number | null> = {}
  await Promise.all(
    symbols.map(async (symbol) => {
      const quote = await getQuote(symbol)
      results[symbol] = extractPrice(quote)
    })
  )
  return results
}

// 取得歷史收盤價

      { headers: { 'X-API-KEY': FUGLE_API_KEY } }
    )
    if (!res.ok) return null
    return res.json()
  } catch {
    return null
  }
}

const GEMINI_API_KEY = import.meta.env.VITE_GEMINI_API_KEY

const MODELS = [
  'gemini-2.5-flash',
  'gemini-2.5-flash-lite',
  'gemini-3-flash-preview',
  'gemini-3.1-flash-lite',
  'gemini-2.0-flash',
  'gemini-1.5-flash',
]

async function callGemini(prompt: string): Promise<string> {
  for (const model of MODELS) {
    try {
      const res = await fetch(
        `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${GEMINI_API_KEY}`,
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{ parts: [{ text: prompt }] }]
          })
        }
      )
      if (!res.ok) {
        const status = res.status
        if (status === 429 || status === 404 || status === 400) {
          console.warn(`Gemini model ${model} failed (${status}), trying next...`)
          continue
        }
        throw new Error(`HTTP ${status}`)
      }
      const data = await res.json()
      const text = data.candidates?.[0]?.content?.parts?.[0]?.text
      if (!text) continue
      console.log(`✅ Gemini using model: ${model}`)
      return text
    } catch (err) {
      console.warn(`Gemini model ${model} error:`, err)
      continue
    }
  }
  throw new Error('所有 Gemini 模型都無法使用')
}

  const [lastUpdated, setLastUpdated] = useState('')
  const [hasPrice, setHasPrice] = useState(false)
  const [upcomingEvents, setUpcomingEvents] = useState<any[]>([])

  const loadData = async () => {
    setLoading(true)
    try {
      const [tradesSnap, cashSnap, priceDoc] = await Promise.all([
        getDocs(query(collection(investDB, 'trades'), orderBy('date', 'asc'))),
        getDocs(collection(investDB, 'cashflow')),
        getDoc(doc(investDB, 'settings', 'lastPrices'))
      ])

      const trades = tradesSnap.docs.map(d => d.data() as TradeRecord)
      const cashRecords = cashSnap.docs.map(d => d.data() as any)
      const { holdings: positions, realized } = calcPositions(trades)
      setRealizedPnL(realized.reduce((s, r) => s + r.realized, 0))

      const savedPrices: Record<string, any> = priceDoc.exists() ? priceDoc.data() : {}
      const savedAt = savedPrices._updatedAt || ''

      let totalMarketValue = 0
      let priceAvailable = false

      const holdingList = Object.values(positions).map(pos => {
        const currentPrice = savedPrices[pos.code] ?? null
        if (currentPrice) priceAvailable = true
        const marketValue = currentPrice ? currentPrice * pos.shares : pos.totalCost
        const unrealizedPnL = currentPrice ? marketValue - pos.totalCost : 0
        const returnRate = pos.totalCost > 0 ? (unrealizedPnL / pos.totalCost) * 100 : 0
        totalMarketValue += marketValue
        return { ...pos, currentPrice, marketValue, unrealizedPnL, returnRate, ratio: 0 }
      })

      holdingList.forEach(h => { h.ratio = totalMarketValue > 0 ? (h.marketValue / totalMarketValue) * 100 : 0 })
      holdingList.sort((a, b) => b.marketValue - a.marketValue)

      setHoldings(holdingList)
      setTotalCash(calcAvailableCash(trades, cashRecords))
      setHasPrice(priceAvailable)
      if (savedAt) setLastUpdated(savedAt)
    } catch (err) { console.error(err) }
    setLoading(false)

function TodoWidget() {
  const [tasks, setTasks] = useState<any[]>([])

  useEffect(() => {
    const load = async () => {
      const snap = await getDocs(query(collection(investDB, 'tasks'), orderBy('order', 'asc')))
      const data = snap.docs.map(d => ({ id: d.id, ...d.data() }))
        .filter((t: any) => !t.done)
        .slice(0, 5)
      setTasks(data)
    }
    load()
  }, [])

  const PRIORITY_COLORS: Record<string, string> = { high: '#ff5757', medium: '#f5a623', low: '#00c896' }

  if (tasks.length === 0) {
    return <div style={{ textAlign: 'center', padding: '20px', color: '#8892a4', fontSize: '12px' }}>🎉 沒有待辦任務</div>
  }

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: '5px' }}>
      {tasks.map((t: any) => {
        const isOverdue = t.dueDate && new Date(t.dueDate) < new Date()
        return (
          <div key={t.id} style={{ display: 'flex', alignItems: 'center', gap: '8px', padding: '6px 10px', backgroundColor: '#0f1117', borderRadius: '7px', borderLeft: `3px solid ${PRIORITY_COLORS[t.priority] || '#4f8ef7'}` }}>
            <div style={{ width: '6px', height: '6px', borderRadius: '50%', backgroundColor: PRIORITY_COLORS[t.priority] || '#4f8ef7', flexShrink: 0 }} />
            <span style={{ fontSize: '12px', color: '#e2e8f0', flex: 1, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{t.title}</span>
            {t.dueDate && (
              <span style={{ fontSize: '10px', color: isOverdue ? '#ff5757' : '#8892a4', flexShrink: 0 }}>
                {isOverdue ? '⚠️' : ''}{t.dueDate.slice(5).replace('-', '/')}
              </span>
            )}
          </div>
        )
      })}
    </div>
  )
}

interface DisplayHolding {
  code: string
  name: string
  shares: number
  avgCost: number
  currentPrice: number | null
  marketValue: number
  totalCost: number
  unrealizedPnL: number
  returnRate: number
}

export default function Holdings() {
  const [holdings, setHoldings] = useState<DisplayHolding[]>([])
  const [loading, setLoading] = useState(true)
  const [lastUpdated, setLastUpdated] = useState('')

  const load = async () => {
    setLoading(true)
    try {
      const snap = await getDocs(query(collection(investDB, 'trades'), orderBy('date', 'asc')))
      const trades = snap.docs.map(d => d.data() as TradeRecord)
      const { holdings: positions } = calcPositions(trades)

      // 讀取 Firebase 儲存的現價（跟儀表板共用）
      const priceDoc = await getDoc(doc(investDB, 'settings', 'lastPrices'))
      const savedPrices: Record<string, any> = priceDoc.exists() ? priceDoc.data() : {}
      const savedAt = savedPrices._updatedAt || ''
      if (savedAt) setLastUpdated(savedAt)

      const list: DisplayHolding[] = Object.values(positions).map(pos => {
        const currentPrice = savedPrices[pos.code] ?? null
        const marketValue = currentPrice ? currentPrice * pos.shares : pos.totalCost
        const unrealizedPnL = currentPrice ? marketValue - pos.totalCost : 0
        const returnRate = pos.totalCost > 0 ? (unrealizedPnL / pos.totalCost) * 100 : 0
        return {
          code: pos.code, name: pos.name, shares: pos.shares,
          avgCost: pos.avgCost, currentPrice, marketValue,
          totalCost: pos.totalCost, unrealizedPnL, returnRate
        }
      }).sort((a, b) => b.marketValue - a.marketValue)

      setHoldings(list)
    } catch (err) { console.error(err) }
    setLoading(false)
  }

  useEffect(() => { load() }, [])

  const totalMarketValue = holdings.reduce((s, h) => s + h.marketValue, 0)
  const totalCost = holdings.reduce((s, h) => s + h.totalCost, 0)
  const totalPnL = holdings.reduce((s, h) => s + h.unrealizedPnL, 0)
  const totalReturn = totalCost > 0 ? (totalPnL / totalCost) * 100 : 0
  const hasPrice = holdings.some(h => h.currentPrice !== null)
  const pnlColor = (v: number) => v >= 0 ? '#00c896' : '#ff5757'
  const card: React.CSSProperties = { backgroundColor: '#1a1d27', border: '1px solid #2e3349', borderRadius: '12px', padding: '20px' }

        .holdings-stats { display: grid; gap: 16px; grid-template-columns: repeat(4, 1fr); margin-bottom: 20px; }

        @media (max-width: 480px) { .holdings-stats { grid-template-columns: 1fr; } }
        .table-wrap { overflow-x: auto; }
        .holdings-table { width: 100%; border-collapse: collapse; font-size: 14px; }
        .holdings-table th, .holdings-table td { padding: 12px 14px; text-align: left; white-space: nowrap; }
        .holdings-table th { color: #8892a4; font-weight: 500; border-bottom: 1px solid #2e3349; font-size: 13px; }
        .holdings-table td { border-bottom: 1px solid #2e3349; color: #e2e8f0; }
        .holdings-table tr:hover td { background-color: rgba(79,142,247,0.04); }
        @keyframes spin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }
      `}</style>

      <div style={{ padding: 'clamp(16px, 3vw, 32px)', maxWidth: '1600px', margin: '0 auto' }}>
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '24px' }}>
          <div>
            <h1 style={{ fontSize: 'clamp(18px, 2.5vw, 22px)', fontWeight: 700, color: '#e2e8f0', margin: 0 }}>股票持倉</h1>

              {lastUpdated ? `現價更新：${lastUpdated}` : '尚未更新現價，請至儀表板點「更新現價」'}
            </p>
          </div>

interface Trade {
  id?: string
  date: string
  type: 'buy' | 'sell'
  code: string
  name: string
  price: number
  shares: number
  fee: number
  tax: number
  total: number
  note: string
}

const btn = (color: string): React.CSSProperties => ({ backgroundColor: color, border: 'none', borderRadius: '8px', padding: '8px 16px', color: '#fff', fontSize: '14px', cursor: 'pointer', display: 'flex', alignItems: 'center', gap: '6px' })

    const handle = () => setIsMobile(window.innerWidth < 768)
    window.addEventListener('resize', handle)
    return () => window.removeEventListener('resize', handle)
  }, [])

  const load = async () => {
    const q = query(collection(investDB, 'trades'), orderBy('date', 'desc'))
    const snap = await getDocs(q)
    setTrades(snap.docs.map(d => ({ id: d.id, ...d.data() } as Trade)))

    setStockDict(dict)
  }

    const upper = code.toUpperCase()
    setForm(f => {
      const name = stockDict[upper] || f.name
      const fee = f.price && f.shares ? calcFee(f.price, f.shares) : f.fee
      const tax = f.type === 'sell' && f.price && f.shares ? calcTax(f.price, f.shares, upper) : 0
      const u = { ...f, code: upper, name, fee, tax }
      u.total = f.type === 'buy' ? f.price * f.shares + u.fee : f.price * f.shares - u.fee - u.tax
      return u
    })
  }

  const handleChange = (key: keyof Trade, value: any) => {

      if (['price', 'shares', 'type', 'code'].includes(key)) {

          u.tax = u.type === 'sell' ? calcTax(u.price, u.shares, u.code) : 0

      const name = (row['名稱'] || '').trim()
      const price = parseFloat(row['價格/金額']) || 0
      const shares = parseInt(row['股數']) || 0
      if (!rawCode || !price || !shares) { failed++; continue }

      const tradeType = type === '買進' ? 'buy' : 'sell'
      const fee = calcFee(price, shares)
      const tax = tradeType === 'sell' ? calcTax(price, shares, rawCode) : 0

        success++
      } catch { failed++ }
    }

    setImportResult({ success, failed, skipped })
    setImporting(false); e.target.value = ''

        @media (max-width: 1100px) { .trades-grid { grid-template-columns: repeat(3, 1fr); } }

          <div>
            <h1 style={{ fontSize: 'clamp(16px, 2.5vw, 22px)', fontWeight: 700, color: '#e2e8f0', margin: 0 }}>買賣紀錄</h1>
            <p style={{ fontSize: '12px', color: '#8892a4', margin: '2px 0 0' }}>共 {trades.length} 筆交易</p>
          </div>
          <div style={{ display: 'flex', gap: '8px' }}>
            <label style={{ ...btn('#21253a'), border: '1px solid #2e3349', cursor: 'pointer', fontSize: '12px', padding: '7px 10px' }}>
              <Upload size={13} />{isMobile ? 'CSV' : importing ? '匯入中...' : '匯入歷史紀錄 CSV'}
              <input ref={fileRef} type="file" accept=".csv" onChange={handleImportCSV} disabled={importing} style={{ display: 'none' }} />
            </label>
            <button style={{ ...btn('#4f8ef7'), fontSize: '12px', padding: '7px 12px' }} onClick={() => { setForm(emptyTrade); setEditingId(null); setShowForm(v => !v) }}>
              {showForm ? <X size={13} /> : <Check size={13} />}{showForm ? '收起' : '新增交易'}
            </button>
          </div>
        </div>

        {importResult && (
          <div style={{ backgroundColor: 'rgba(0,200,150,0.08)', border: '1px solid rgba(0,200,150,0.25)', borderRadius: '10px', padding: '12px 16px', marginBottom: '12px', display: 'flex', gap: '20px', flexWrap: 'wrap' }}>
            <div><span style={{ fontSize: '11px', color: '#8892a4' }}>成功匯入</span><div style={{ fontSize: '16px', fontWeight: 700, color: '#00c896' }}>{importResult.success} 筆</div></div>
            {importResult.failed > 0 && <div><span style={{ fontSize: '11px', color: '#8892a4' }}>失敗</span><div style={{ fontSize: '16px', fontWeight: 700, color: '#ff5757' }}>{importResult.failed} 筆</div></div>}
            {importResult.skipped > 0 && <div><span style={{ fontSize: '11px', color: '#8892a4' }}>略過</span><div style={{ fontSize: '16px', fontWeight: 700, color: '#8892a4' }}>{importResult.skipped} 筆</div></div>}
          </div>
        )}

        {/* 表單 */}
        {showForm && (
          <div style={{ ...card, marginBottom: '16px' }}>
            <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '16px' }}>
              {editingId ? '✏️ 編輯交易' : '➕ 新增交易'}
            </div>
            <div className="trades-grid">
              <div><label style={lbl}>日期</label><input type="date" style={inp} value={form.date} onChange={e => handleChange('date', e.target.value)} /></div>
              <div><label style={lbl}>類型</label>
                <select style={inp} value={form.type} onChange={e => handleChange('type', e.target.value)}>
                  <option value="buy">買入</option>
                  <option value="sell">賣出</option>
                </select>
              </div>
              <div><label style={lbl}>股票代號</label><input style={inp} placeholder="例：2330" value={form.code} onChange={e => handleCodeChange(e.target.value)} /></div>
              <div><label style={lbl}>股票名稱</label><input style={inp} placeholder="例：台積電" value={form.name} onChange={e => handleChange('name', e.target.value)} /></div>
              <div><label style={lbl}>價格</label><input type="number" style={inp} placeholder="0" value={form.price || ''} onChange={e => handleChange('price', parseFloat(e.target.value) || 0)} /></div>
              <div><label style={lbl}>股數</label><input type="number" style={inp} placeholder="0" value={form.shares || ''} onChange={e => handleChange('shares', parseInt(e.target.value) || 0)} /></div>
              <div>
                <label style={lbl}>手續費 <span style={{ color: '#4f8ef7', fontSize: '11px' }}>（6折自動計算）</span></label>
                <input type="number" style={inp} value={form.fee || ''} onChange={e => handleChange('fee', parseFloat(e.target.value) || 0)} />
              </div>
              <div>
                <label style={lbl}>交易稅 <span style={{ color: form.type === 'sell' ? (isETF(form.code) ? '#f5a623' : '#4f8ef7') : '#2e3349', fontSize: '11px' }}>
                  {form.type === 'sell' ? (isETF(form.code) ? '（ETF千分之一）' : '（千分之三）') : '（買入免稅）'}
                </span></label>
                <input type="number" style={inp} value={form.tax || ''} onChange={e => handleChange('tax', parseFloat(e.target.value) || 0)} />
              </div>
              <div><label style={lbl}>收付金額</label><input type="number" style={{ ...inp, color: form.type === 'buy' ? '#ff5757' : '#00c896' }} value={form.total} readOnly /></div>
              <div className="span2"><label style={lbl}>備註</label><input style={inp} placeholder="選填" value={form.note} onChange={e => handleChange('note', e.target.value)} /></div>
            </div>
            <div style={{ display: 'flex', gap: '8px', marginTop: '16px', justifyContent: 'flex-end' }}>
              {editingId && <button style={btn('#2e3349')} onClick={handleCancel}><X size={14} />取消</button>}
              <button style={btn('#4f8ef7')} onClick={handleSave} disabled={loading}>
                <Check size={14} />{loading ? '儲存中...' : editingId ? '確認更新' : '確認新增'}
              </button>
            </div>
          </div>
        )}

        {/* 搜尋列 */}
        <div style={{ marginBottom: '12px', position: 'relative' }}>
          <Search size={14} style={{ position: 'absolute', left: '12px', top: '50%', transform: 'translateY(-50%)', color: '#8892a4' }} />
          <input style={{ ...inp, paddingLeft: '36px', backgroundColor: '#1a1d27', border: '1px solid #2e3349' }}
            placeholder="搜尋代號或名稱..." value={search} onChange={e => setSearch(e.target.value)} />
        </div>

        {/* 列表 */}
        <div style={card}>
          {filtered.length === 0 ? (
            <div style={{ textAlign: 'center', padding: '48px', color: '#8892a4' }}>
              {search ? `找不到「${search}」相關紀錄` : '還沒有交易紀錄'}
            </div>
          ) : isMobile ? (
            // 手機版：卡片式顯示
            <div>
              {filtered.map(trade => (
                <div key={trade.id} className="trade-card">
                  <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}
                    onClick={() => setExpandedId(expandedId === trade.id ? null : trade.id!)}>
                    <div style={{ display: 'flex', alignItems: 'center', gap: '10px', flex: 1, minWidth: 0 }}>
                      <span style={{ backgroundColor: trade.type === 'buy' ? 'rgba(255,87,87,0.15)' : 'rgba(0,200,150,0.15)', color: trade.type === 'buy' ? '#ff5757' : '#00c896', padding: '2px 8px', borderRadius: '20px', fontSize: '11px', fontWeight: 600, flexShrink: 0 }}>
                        {trade.type === 'buy' ? '買' : '賣'}
                      </span>
                      <span style={{ color: '#4f8ef7', fontWeight: 700, fontSize: '14px', flexShrink: 0 }}>{trade.code}</span>
                      <span style={{ color: '#e2e8f0', fontSize: '13px', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{trade.name}</span>
                    </div>
                    <div style={{ display: 'flex', alignItems: 'center', gap: '8px', flexShrink: 0 }}>
                      <span style={{ color: trade.type === 'buy' ? '#ff5757' : '#00c896', fontWeight: 600, fontSize: '13px' }}>
                        ${trade.total.toLocaleString()}
                      </span>
                      {expandedId === trade.id ? <ChevronUp size={14} color="#8892a4" /> : <ChevronDown size={14} color="#8892a4" />}
                    </div>
                  </div>

                  {expandedId === trade.id && (
                    <div style={{ marginTop: '10px', paddingTop: '10px', borderTop: '1px solid #2e3349' }}>
                      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '6px', marginBottom: '10px' }}>
                        {[
                          { label: '日期', value: trade.date },
                          { label: '價格', value: trade.price.toLocaleString() },
                          { label: '股數', value: trade.shares.toLocaleString() },
                          { label: '手續費', value: trade.fee.toLocaleString() },
                          { label: '交易稅', value: trade.tax.toLocaleString() },
                          { label: '備註', value: trade.note || '—' },
                        ].map(({ label, value }) => (
                          <div key={label}>
                            <div style={{ fontSize: '10px', color: '#8892a4' }}>{label}</div>
                            <div style={{ fontSize: '13px', color: '#e2e8f0' }}>{value}</div>
                          </div>
                        ))}
                      </div>
                      <div style={{ display: 'flex', gap: '8px', justifyContent: 'flex-end' }}>
                        <button onClick={() => handleEdit(trade)} style={{ ...btn('#21253a'), border: '1px solid #2e3349', fontSize: '12px', padding: '6px 12px' }}>
                          <Pencil size={12} />編輯
                        </button>
                        <button onClick={() => handleDelete(trade.id!)} style={{ ...btn('rgba(255,87,87,0.15)'), color: '#ff5757', fontSize: '12px', padding: '6px 12px' }}>
                          <Trash2 size={12} />刪除
                        </button>
                      </div>
                    </div>
                  )}
                </div>
              ))}
            </div>
          ) : (
            // 桌面版：表格顯示
            <div style={{ overflowX: 'auto' }}>
              <table className="trade-table">
                <thead>
                  <tr>{['日期', '類型', '代號', '名稱', '價格', '股數', '手續費', '交易稅', '收付金額', '備註', ''].map(h => <th key={h}>{h}</th>)}</tr>
                </thead>
                <tbody>
                  {filtered.map(trade => (
                    <tr key={trade.id} style={{ backgroundColor: editingId === trade.id ? 'rgba(79,142,247,0.06)' : 'transparent' }}>
                      <td>{trade.date}</td>
                      <td><span style={{ backgroundColor: trade.type === 'buy' ? 'rgba(255,87,87,0.15)' : 'rgba(0,200,150,0.15)', color: trade.type === 'buy' ? '#ff5757' : '#00c896', padding: '2px 10px', borderRadius: '20px', fontSize: '12px', fontWeight: 600 }}>{trade.type === 'buy' ? '買入' : '賣出'}</span></td>
                      <td style={{ color: '#4f8ef7', fontWeight: 700 }}>{trade.code}</td>
                      <td>{trade.name}</td>
                      <td>{trade.price.toLocaleString()}</td>
                      <td>{trade.shares.toLocaleString()}</td>
                      <td>{trade.fee.toLocaleString()}</td>
                      <td>{trade.tax.toLocaleString()}</td>
                      <td style={{ color: trade.type === 'buy' ? '#ff5757' : '#00c896', fontWeight: 600 }}>{trade.total.toLocaleString()}</td>
                      <td style={{ color: '#8892a4' }}>{trade.note}</td>
                      <td>
                        <div style={{ display: 'flex', gap: '8px' }}>
                          <button onClick={() => handleEdit(trade)} style={{ background: 'none', border: 'none', color: '#4f8ef7', cursor: 'pointer' }}><Pencil size={15} /></button>
                          <button onClick={() => handleDelete(trade.id!)} style={{ background: 'none', border: 'none', color: '#ff5757', cursor: 'pointer' }}><Trash2 size={15} /></button>
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </div>
      </div>
    </>
  )
}

import { useState, useEffect } from 'react'
import { collection, getDocs, setDoc, deleteDoc, doc, updateDoc } from 'firebase/firestore'
import { investDB } from '../../lib/firebase'
import { getMultipleQuotes } from '../../lib/fugle'
import { Plus, Trash2, RefreshCw, TrendingUp, TrendingDown, Pencil, Check, X, FileText } from 'lucide-react'

interface WatchItem {
  code: string
  name: string
  targetHigh?: number
  targetLow?: number
  note?: string
  currentPrice?: number
  addedAt: string
}

const card: React.CSSProperties = { backgroundColor: '#1a1d27', border: '1px solid #2e3349', borderRadius: '12px', padding: '20px' }
const inp: React.CSSProperties = { backgroundColor: '#0f1117', border: '1px solid #2e3349', borderRadius: '8px', padding: '8px 12px', color: '#e2e8f0', fontSize: '14px', width: '100%', outline: 'none', boxSizing: 'border-box' }
const lbl: React.CSSProperties = { fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }
const btn = (color: string): React.CSSProperties => ({ backgroundColor: color, border: 'none', borderRadius: '8px', padding: '8px 16px', color: '#fff', fontSize: '14px', cursor: 'pointer', display: 'flex', alignItems: 'center', gap: '6px' })

export default function Watchlist() {
  const [items, setItems] = useState<WatchItem[]>([])
  const [loading, setLoading] = useState(true)
  const [priceLoading, setPriceLoading] = useState(false)
  const [showForm, setShowForm] = useState(false)
  const [editingCode, setEditingCode] = useState<string | null>(null)
  const [expandedNote, setExpandedNote] = useState<string | null>(null)
  const [stockDict, setStockDict] = useState<Record<string, string>>({})
  const [form, setForm] = useState({ code: '', name: '', targetHigh: '', targetLow: '', note: '' })
  const [lastUpdated, setLastUpdated] = useState('')

  const load = async () => {
    setLoading(true)
    const snap = await getDocs(collection(investDB, 'watchlist'))
    const data = snap.docs.map(d => ({ code: d.id, ...d.data() } as WatchItem))
    // 讀取儲存的現價
    const priceDoc = await doc(investDB, 'settings', 'watchlistPrices')
    const priceSnap = await getDocs(collection(investDB, 'settings'))
    const priceData = priceSnap.docs.find(d => d.id === 'watchlistPrices')?.data() || {}
    const savedAt = priceData._updatedAt || ''
    if (savedAt) setLastUpdated(savedAt)
    data.forEach(item => { item.currentPrice = priceData[item.code] ?? undefined })
    setItems(data.sort((a, b) => a.code.localeCompare(b.code)))

    // 讀取股票字典
    const dictSnap = await getDocs(collection(investDB, 'stockDict'))
    const dict: Record<string, string> = {}
    dictSnap.docs.forEach(d => { dict[d.id] = d.data().name })
    setStockDict(dict)
    setLoading(false)
  }

  useEffect(() => { load() }, [])

  // 更新現價
  const updatePrices = async () => {
    if (items.length === 0) return
    setPriceLoading(true)
    try {
      const codes = items.map(i => i.code)
      const quotes = await getMultipleQuotes(codes)
      const now = new Date().toLocaleString('zh-TW')
      const priceData: Record<string, any> = { _updatedAt: now }
      codes.forEach(c => { if (quotes[c] !== null) priceData[c] = quotes[c] })
      await setDoc(doc(investDB, 'settings', 'watchlistPrices'), priceData)
      setItems(prev => prev.map(item => ({ ...item, currentPrice: quotes[item.code] ?? item.currentPrice })))
      setLastUpdated(now)
    } catch (err) { console.error(err) }
    setPriceLoading(false)
  }

  const handleAdd = async () => {
    if (!form.code.trim()) return
    const code = form.code.trim()
    const name = form.name.trim() || stockDict[code] || code
    await setDoc(doc(investDB, 'watchlist', code), {
      name,
      targetHigh: form.targetHigh ? parseFloat(form.targetHigh) : null,
      targetLow: form.targetLow ? parseFloat(form.targetLow) : null,
      note: form.note || null,
      addedAt: new Date().toISOString()
    })
    setForm({ code: '', name: '', targetHigh: '', targetLow: '', note: '' })
    setShowForm(false); load()
  }

  const handleEdit = async (code: string) => {
    const item = items.find(i => i.code === code)
    if (!item) return
    await updateDoc(doc(investDB, 'watchlist', code), {
      targetHigh: form.targetHigh ? parseFloat(form.targetHigh) : null,
      targetLow: form.targetLow ? parseFloat(form.targetLow) : null,
      note: form.note || null,
      name: form.name || item.name
    })
    setEditingCode(null); load()
  }

  const handleDelete = async (code: string) => {
    if (!confirm(`確定要從自選股移除 ${code}？`)) return
    await deleteDoc(doc(investDB, 'watchlist', code)); load()
  }

  const handleCodeChange = (code: string) => {
    setForm(f => ({ ...f, code, name: stockDict[code] || f.name }))
  }

  const getPriceStatus = (item: WatchItem) => {
    if (!item.currentPrice) return 'normal'
    if (item.targetHigh && item.currentPrice >= item.targetHigh) return 'high'
    if (item.targetLow && item.currentPrice <= item.targetLow) return 'low'
    return 'normal'
  }

  const getDistancePct = (current: number, target: number) =>
    (((target - current) / current) * 100).toFixed(1)

  return (
    <>
      <style>{`
        .watch-grid { display: grid; gap: 16px; grid-template-columns: repeat(auto-fill, minmax(340px, 1fr)); }
        @media (max-width: 700px) { .watch-grid { grid-template-columns: 1fr; } }
        .form-grid { display: grid; gap: 12px; grid-template-columns: 1fr 1fr; }
        @media (max-width: 600px) { .form-grid { grid-template-columns: 1fr; } }
        @keyframes spin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }
        .watch-card { transition: border-color 0.2s; }
        .watch-card:hover { border-color: #4f8ef7 !important; }
      `}</style>

      <div style={{ padding: 'clamp(16px, 3vw, 32px)', maxWidth: '1400px', margin: '0 auto' }}>
        {/* 標題列 */}
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '24px', flexWrap: 'wrap', gap: '12px' }}>
          <div>
            <h1 style={{ fontSize: 'clamp(18px, 2.5vw, 22px)', fontWeight: 700, color: '#e2e8f0', margin: 0 }}>自選股追蹤</h1>
            <p style={{ fontSize: '13px', color: '#8892a4', margin: '4px 0 0' }}>
              {lastUpdated ? `現價更新：${lastUpdated}` : '點「更新現價」取得最新報價'}
            </p>
          </div>
          <div style={{ display: 'flex', gap: '8px' }}>
            <button onClick={updatePrices} disabled={priceLoading || items.length === 0} style={btn('#21253a')}>
              <RefreshCw size={14} style={{ animation: priceLoading ? 'spin 1s linear infinite' : 'none' }} />
              {priceLoading ? '更新中...' : '更新現價'}
            </button>
            <button onClick={() => { setShowForm(true) }} style={btn('#4f8ef7')}>
              <Plus size={14} />新增自選股
            </button>
          </div>
        </div>

        {/* 新增表單 */}
        {showForm && (
          <div style={{ ...card, marginBottom: '24px' }}>
            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '16px' }}>
              <span style={{ fontSize: '15px', fontWeight: 600, color: '#e2e8f0' }}>➕ 新增自選股</span>
              <button onClick={() => setShowForm(false)} style={{ background: 'none', border: 'none', color: '#8892a4', cursor: 'pointer' }}><X size={18} /></button>
            </div>
            <div className="form-grid">
              <div>
                <label style={lbl}>股票代號 *</label>
                <input style={inp} placeholder="例如：2330" value={form.code}
                  onChange={e => handleCodeChange(e.target.value)} />
              </div>
              <div>
                <label style={lbl}>股票名稱</label>
                <input style={inp} placeholder="自動帶入" value={form.name}
                  onChange={e => setForm(f => ({ ...f, name: e.target.value }))} />
              </div>
              <div>
                <label style={lbl}>目標上標價（賣出參考）</label>
                <input type="number" style={inp} placeholder="0" value={form.targetHigh}
                  onChange={e => setForm(f => ({ ...f, targetHigh: e.target.value }))} />
              </div>
              <div>
                <label style={lbl}>目標下標價（買入參考）</label>
                <input type="number" style={inp} placeholder="0" value={form.targetLow}
                  onChange={e => setForm(f => ({ ...f, targetLow: e.target.value }))} />
              </div>
              <div style={{ gridColumn: 'span 2' }}>
                <label style={lbl}>個人評估筆記</label>
                <textarea style={{ ...inp, minHeight: '80px', resize: 'vertical', fontFamily: 'sans-serif' }}
                  placeholder="記錄這支股票的評估理由、觀察重點..."
                  value={form.note} onChange={e => setForm(f => ({ ...f, note: e.target.value }))} />
              </div>
            </div>
            <div style={{ display: 'flex', gap: '8px', marginTop: '16px', justifyContent: 'flex-end' }}>
              <button style={btn('#2e3349')} onClick={() => setShowForm(false)}><X size={14} />取消</button>
              <button style={btn('#4f8ef7')} onClick={handleAdd}><Check size={14} />新增</button>
            </div>
          </div>
        )}

        {/* 自選股列表 */}
        {loading ? (
          <div style={{ textAlign: 'center', padding: '64px', color: '#8892a4' }}>載入中...</div>
        ) : items.length === 0 ? (
          <div style={{ textAlign: 'center', padding: '64px', color: '#8892a4' }}>
            <p style={{ fontSize: '15px' }}>還沒有自選股</p>
            <p style={{ fontSize: '13px', marginTop: '8px' }}>點「新增自選股」開始追蹤</p>
          </div>
        ) : (
          <div className="watch-grid">
            {items.map(item => {
              const status = getPriceStatus(item)
              const isEditing = editingCode === item.code
              const borderColor = status === 'high' ? '#00c896' : status === 'low' ? '#ff5757' : '#2e3349'

              return (
                <div key={item.code} className="watch-card"
                  style={{ ...card, border: `1px solid ${borderColor}`, display: 'flex', flexDirection: 'column', gap: '12px' }}>

                  {/* 卡片頭 */}
                  <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
                    <div>
                      <div style={{ display: 'flex', alignItems: 'center', gap: '8px' }}>
                        <span style={{ fontSize: '18px', fontWeight: 700, color: '#4f8ef7' }}>{item.code}</span>
                        {status === 'high' && <span style={{ fontSize: '11px', backgroundColor: 'rgba(0,200,150,0.15)', color: '#00c896', borderRadius: '10px', padding: '2px 8px' }}>🎯 達到上標</span>}
                        {status === 'low' && <span style={{ fontSize: '11px', backgroundColor: 'rgba(255,87,87,0.15)', color: '#ff5757', borderRadius: '10px', padding: '2px 8px' }}>🎯 達到下標</span>}
                      </div>
                      <div style={{ fontSize: '14px', color: '#e2e8f0', marginTop: '2px' }}>{item.name}</div>
                    </div>
                    <div style={{ display: 'flex', gap: '6px' }}>
                      <button onClick={() => {
                        setEditingCode(item.code)
                        setForm({ code: item.code, name: item.name, targetHigh: item.targetHigh?.toString() || '', targetLow: item.targetLow?.toString() || '', note: item.note || '' })
                      }} style={{ background: 'none', border: 'none', color: '#8892a4', cursor: 'pointer' }}><Pencil size={14} /></button>
                      <button onClick={() => handleDelete(item.code)} style={{ background: 'none', border: 'none', color: '#8892a4', cursor: 'pointer' }}><Trash2 size={14} /></button>
                    </div>
                  </div>

                  {/* 現價 */}
                  <div style={{ display: 'flex', gap: '16px', alignItems: 'center' }}>
                    <div>
                      <div style={{ fontSize: '11px', color: '#8892a4', marginBottom: '2px' }}>現價</div>
                      <div style={{ fontSize: '22px', fontWeight: 700, color: '#e2e8f0' }}>
                        {item.currentPrice ? item.currentPrice.toFixed(2) : '—'}
                      </div>
                    </div>
                    {item.currentPrice && item.targetHigh && (
                      <div style={{ flex: 1 }}>

                        </div>
                      </div>
                    )}
                  </div>

                  {/* 目標價 */}
                  {isEditing ? (
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '8px' }}>
                      <div style={{ display: 'flex', gap: '8px' }}>
                        <div style={{ flex: 1 }}>
                          <label style={lbl}>名稱</label>
                          <input style={inp} value={form.name} onChange={e => setForm(f => ({ ...f, name: e.target.value }))} />
                        </div>
                      </div>
                      <div style={{ display: 'flex', gap: '8px' }}>
                        <div style={{ flex: 1 }}>
                          <label style={lbl}>上標價</label>
                          <input type="number" style={inp} placeholder="賣出參考" value={form.targetHigh}
                            onChange={e => setForm(f => ({ ...f, targetHigh: e.target.value }))} />
                        </div>
                        <div style={{ flex: 1 }}>
                          <label style={lbl}>下標價</label>
                          <input type="number" style={inp} placeholder="買入參考" value={form.targetLow}
                            onChange={e => setForm(f => ({ ...f, targetLow: e.target.value }))} />
                        </div>
                      </div>
                      <div>
                        <label style={lbl}>評估筆記</label>
                        <textarea style={{ ...inp, minHeight: '60px', resize: 'vertical', fontFamily: 'sans-serif' }}
                          value={form.note} onChange={e => setForm(f => ({ ...f, note: e.target.value }))} />
                      </div>
                      <div style={{ display: 'flex', gap: '8px', justifyContent: 'flex-end' }}>
                        <button style={{ ...btn('#2e3349'), fontSize: '12px', padding: '6px 12px' }} onClick={() => setEditingCode(null)}><X size={12} />取消</button>
                        <button style={{ ...btn('#4f8ef7'), fontSize: '12px', padding: '6px 12px' }} onClick={() => handleEdit(item.code)}><Check size={12} />儲存</button>
                      </div>
                    </div>
                  ) : (
                    <>
                      {/* 目標價顯示 */}
                      <div style={{ display: 'flex', gap: '12px' }}>
                        {item.targetLow && (
                          <div style={{ flex: 1, backgroundColor: 'rgba(255,87,87,0.08)', borderRadius: '8px', padding: '8px 12px' }}>
                            <div style={{ fontSize: '11px', color: '#8892a4', marginBottom: '2px' }}>下標（買入）</div>
                            <div style={{ fontSize: '15px', fontWeight: 600, color: '#ff5757' }}>
                              {item.targetLow}
                              {item.currentPrice && (
                                <span style={{ fontSize: '11px', color: '#8892a4', marginLeft: '6px' }}>
                                  ({getDistancePct(item.currentPrice, item.targetLow)}%)
                                </span>
                              )}
                            </div>
                          </div>
                        )}
                        {item.targetHigh && (
                          <div style={{ flex: 1, backgroundColor: 'rgba(0,200,150,0.08)', borderRadius: '8px', padding: '8px 12px' }}>
                            <div style={{ fontSize: '11px', color: '#8892a4', marginBottom: '2px' }}>上標（賣出）</div>
                            <div style={{ fontSize: '15px', fontWeight: 600, color: '#00c896' }}>
                              {item.targetHigh}
                              {item.currentPrice && (
                                <span style={{ fontSize: '11px', color: '#8892a4', marginLeft: '6px' }}>
                                  ({getDistancePct(item.currentPrice, item.targetHigh)}%)
                                </span>
                              )}
                            </div>
                          </div>
                        )}
                      </div>

                      {/* 評估筆記 */}
                      {item.note && (
                        <div style={{ borderTop: '1px solid #2e3349', paddingTop: '10px' }}>
                          <button onClick={() => setExpandedNote(expandedNote === item.code ? null : item.code)}
                            style={{ display: 'flex', alignItems: 'center', gap: '6px', background: 'none', border: 'none', color: '#8892a4', cursor: 'pointer', fontSize: '12px', padding: 0, marginBottom: '6px' }}>
                            <FileText size={12} />評估筆記
                          </button>
                          {expandedNote === item.code && (
                            <p style={{ fontSize: '12px', color: '#8892a4', margin: 0, lineHeight: 1.6, whiteSpace: 'pre-wrap' }}>{item.note}</p>
                          )}
                        </div>
                      )}
                    </>
                  )}

                  <div style={{ fontSize: '11px', color: '#8892a4', borderTop: '1px solid #2e3349', paddingTop: '8px' }}>
                    加入於 {new Date(item.addedAt).toLocaleDateString('zh-TW')}
                  </div>
                </div>
              )
            })}
          </div>
        )}
      </div>
    </>
  )
}

import { useState, useEffect } from 'react'
import { collection, getDocs, orderBy, query } from 'firebase/firestore'
import { investDB } from '../../lib/firebase'
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Cell, LineChart, Line, Legend } from 'recharts'
import { calcPositions, calcAvailableCash, type TradeRecord } from '../../lib/calc'

interface MonthlyPnL {
  month: string
  realized: number
  dividend: number
  total: number
}

interface YearlyPerf {
  year: number
  startValue: number
  endValue: number
  invested: number
  cagr: number
  totalReturn: number
}

const card: React.CSSProperties = { backgroundColor: '#1a1d27', border: '1px solid #2e3349', borderRadius: '12px', padding: '20px' }
const inp: React.CSSProperties = { backgroundColor: '#0f1117', border: '1px solid #2e3349', borderRadius: '8px', padding: '8px 12px', color: '#e2e8f0', fontSize: '14px', outline: 'none' }
const pnlColor = (v: number) => v >= 0 ? '#00c896' : '#ff5757'

const CustomTooltip = ({ active, payload, label }: any) => {
  if (!active || !payload?.length) return null
  return (
    <div style={{ backgroundColor: '#21253a', border: '1px solid #2e3349', borderRadius: '8px', padding: '12px 16px' }}>
      <p style={{ color: '#8892a4', fontSize: '12px', margin: '0 0 8px' }}>{label}</p>
      {payload.map((p: any) => (
        <p key={p.name} style={{ color: p.color, fontSize: '14px', fontWeight: 600, margin: '2px 0' }}>
          {p.name}：{typeof p.value === 'number' && p.name.includes('%') ? `${p.value.toFixed(2)}%` : `$${Number(p.value).toLocaleString()}`}
        </p>
      ))}
    </div>
  )
}

// 計算 CAGR
function calcCAGR(startValue: number, endValue: number, years: number): number {
  if (startValue <= 0 || years <= 0) return 0
  return (Math.pow(endValue / startValue, 1 / years) - 1) * 100
}

export default function YearReview() {
  const [monthlyData, setMonthlyData] = useState<MonthlyPnL[]>([])
  const [selectedYear, setSelectedYear] = useState(new Date().getFullYear())
  const [startMonth, setStartMonth] = useState(1)
  const [endMonth, setEndMonth] = useState(12)
  const [loading, setLoading] = useState(true)
  const [years, setYears] = useState<number[]>([])

  // 績效報告 state
  const [yearlyPerf, setYearlyPerf] = useState<YearlyPerf[]>([])
  const [customStart, setCustomStart] = useState('')
  const [customEnd, setCustomEnd] = useState('')
  const [customCAGR, setCustomCAGR] = useState<number | null>(null)
  const [assetSnapshots, setAssetSnapshots] = useState<Record<string, number>>({})
  const [perfLoading, setPerfLoading] = useState(false)

  const load = async () => {
    setLoading(true)
    try {
      const [tradesSnap, dividendsSnap, snapshotsSnap] = await Promise.all([
        getDocs(query(collection(investDB, 'trades'), orderBy('date', 'asc'))),
        getDocs(query(collection(investDB, 'dividends'), orderBy('date', 'asc'))),
        getDocs(query(collection(investDB, 'assetSnapshots'), orderBy('date', 'asc')))
      ])

      const trades = tradesSnap.docs.map(d => d.data())
      const dividends = dividendsSnap.docs.map(d => d.data())

      // 資產快照
      const snapshots: Record<string, number> = {}
      snapshotsSnap.docs.forEach(d => { snapshots[d.data().date] = d.data().totalAssets })
      setAssetSnapshots(snapshots)

      // 取得所有年份
      const allDates = [...trades.map((t: any) => t.date), ...dividends.map((d: any) => d.date)]
      const allYears = [...new Set(allDates.map((d: string) => parseInt(d.slice(0, 4))))].sort()
      if (allYears.length > 0) {
        setYears(allYears)
        // 預設自訂區間
        if (!customStart) setCustomStart(`${allYears[0]}-01-01`)
        if (!customEnd) setCustomEnd(new Date().toISOString().split('T')[0])
      }

      // 計算每月已實現損益
      const map: Record<string, { shares: number; cost: number }> = {}
      const monthlyRealized: Record<string, number> = {}
      const monthlyDividend: Record<string, number> = {}

      trades.forEach((t: any) => {
        const code = t.code
        if (!map[code]) map[code] = { shares: 0, cost: 0 }
        if (t.type === 'buy') {
          map[code].shares += t.shares
          map[code].cost += t.price * t.shares + t.fee
        } else {
          const avg = map[code].shares > 0 ? map[code].cost / map[code].shares : 0
          const pnl = (t.price - avg) * t.shares - t.fee - t.tax
          const month = t.date.slice(0, 7)
          monthlyRealized[month] = (monthlyRealized[month] || 0) + pnl
          map[code].shares -= t.shares
          map[code].cost -= avg * t.shares
        }
      })

      dividends.forEach((d: any) => {
        const month = d.date.slice(0, 7)
        monthlyDividend[month] = (monthlyDividend[month] || 0) + d.cashDividend
      })

      const result: MonthlyPnL[] = []
      for (let m = 1; m <= 12; m++) {
        const month = `${selectedYear}-${String(m).padStart(2, '0')}`
        const realized = monthlyRealized[month] || 0
        const dividend = monthlyDividend[month] || 0
        result.push({ month: `${m}月`, realized, dividend, total: realized + dividend })
      }
      setMonthlyData(result)

      // 計算年度績效
      calcYearlyPerf(allYears, snapshots, trades as TradeRecord[])
    } catch (err) { console.error(err) }
    setLoading(false)
  }

  const calcYearlyPerf = (allYears: number[], snapshots: Record<string, number>, trades: TradeRecord[]) => {
    const perf: YearlyPerf[] = []

    allYears.forEach(year => {
      // 找該年度最接近年初和年末的快照
      const yearDates = Object.keys(snapshots).filter(d => d.startsWith(`${year}`)).sort()
      const prevYearDates = Object.keys(snapshots).filter(d => d.startsWith(`${year - 1}`)).sort()

      const startDate = prevYearDates[prevYearDates.length - 1] || yearDates[0]
      const endDate = yearDates[yearDates.length - 1]

      if (!startDate || !endDate) return

      const startValue = snapshots[startDate] || 0
      const endValue = snapshots[endDate] || 0

      // 該年度淨投入
      const yearTrades = trades.filter(t => t.date.startsWith(`${year}`))
      const invested = yearTrades.reduce((s, t) => {
        if (t.type === 'buy') return s + t.price * t.shares + (t.fee || 0)
        return s - t.price * t.shares + (t.fee || 0) + (t.tax || 0)
      }, 0)

      const totalReturn = startValue > 0 ? ((endValue - startValue) / startValue) * 100 : 0
      const cagr = calcCAGR(startValue, endValue, 1)

      perf.push({ year, startValue, endValue, invested, cagr, totalReturn })
    })

    setYearlyPerf(perf)
  }

  // 計算自訂區間 CAGR
  const calcCustomCAGR = () => {
    if (!customStart || !customEnd) return
    setPerfLoading(true)

    const startDates = Object.keys(assetSnapshots).filter(d => d >= customStart).sort()
    const endDates = Object.keys(assetSnapshots).filter(d => d <= customEnd).sort()

    const startDate = startDates[0]
    const endDate = endDates[endDates.length - 1]

    if (!startDate || !endDate) { setCustomCAGR(null); setPerfLoading(false); return }

    const startValue = assetSnapshots[startDate]
    const endValue = assetSnapshots[endDate]

    const startMs = new Date(startDate).getTime()
    const endMs = new Date(endDate).getTime()
    const years = (endMs - startMs) / (365.25 * 24 * 60 * 60 * 1000)

    setCustomCAGR(calcCAGR(startValue, endValue, years))
    setPerfLoading(false)
  }

  useEffect(() => { load() }, [selectedYear])

  const filtered = monthlyData.slice(startMonth - 1, endMonth)
  const totalRealized = filtered.reduce((s, m) => s + m.realized, 0)
  const totalDividend = filtered.reduce((s, m) => s + m.dividend, 0)
  const totalPnL = totalRealized + totalDividend
  const months = Array.from({ length: 12 }, (_, i) => i + 1)

  // 整體 CAGR（第一筆到最新）
  const overallCAGR = (() => {
    const dates = Object.keys(assetSnapshots).sort()
    if (dates.length < 2) return null
    const startValue = assetSnapshots[dates[0]]
    const endValue = assetSnapshots[dates[dates.length - 1]]
    const years = (new Date(dates[dates.length - 1]).getTime() - new Date(dates[0]).getTime()) / (365.25 * 24 * 60 * 60 * 1000)
    return calcCAGR(startValue, endValue, years)
  })()

  const perfChartData = yearlyPerf.map(p => ({
    year: `${p.year}`,
    '年度報酬率%': parseFloat(p.totalReturn.toFixed(2)),
    '年末資產': p.endValue
  }))

  return (
    <>
      <style>{`
        .review-stats { display: grid; gap: 16px; grid-template-columns: repeat(3, 1fr); margin-bottom: 20px; }
        @media (max-width: 700px) { .review-stats { grid-template-columns: 1fr; } }
        .perf-stats { display: grid; gap: 16px; grid-template-columns: repeat(4, 1fr); margin-bottom: 20px; }
        @media (max-width: 900px) { .perf-stats { grid-template-columns: repeat(2, 1fr); } }
        @media (max-width: 480px) { .perf-stats { grid-template-columns: 1fr; } }
        .perf-table { width: 100%; border-collapse: collapse; font-size: 13px; }
        .perf-table th { padding: 10px 12px; text-align: left; color: #8892a4; font-weight: 500; border-bottom: 1px solid #2e3349; white-space: nowrap; }
        .perf-table td { padding: 12px; border-bottom: 1px solid #2e3349; color: #e2e8f0; white-space: nowrap; }
      `}</style>

      <div style={{ padding: 'clamp(16px, 3vw, 32px)', maxWidth: '1600px', margin: '0 auto' }}>

        {/* ===== 年度復盤（原有功能）===== */}
        <div style={{ marginBottom: '24px' }}>
          <h1 style={{ fontSize: 'clamp(18px, 2.5vw, 22px)', fontWeight: 700, color: '#e2e8f0', margin: 0 }}>年度復盤</h1>
          <p style={{ fontSize: '13px', color: '#8892a4', margin: '4px 0 0' }}>已實現損益與股利收入分析</p>
        </div>

        {/* 篩選列 */}
        <div style={{ ...card, marginBottom: '20px', display: 'flex', gap: '16px', flexWrap: 'wrap', alignItems: 'flex-end' }}>
          <div>
            <label style={{ fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }}>年度</label>
            <select style={inp} value={selectedYear} onChange={e => setSelectedYear(parseInt(e.target.value))}>
              {years.length === 0
                ? <option value={new Date().getFullYear()}>{new Date().getFullYear()}</option>
                : years.map(y => <option key={y} value={y}>{y}</option>)
              }
            </select>
          </div>
          <div>
            <label style={{ fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }}>起始月份</label>
            <select style={inp} value={startMonth} onChange={e => setStartMonth(parseInt(e.target.value))}>
              {months.map(m => <option key={m} value={m}>{m} 月</option>)}
            </select>
          </div>
          <div>
            <label style={{ fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }}>結束月份</label>
            <select style={inp} value={endMonth} onChange={e => setEndMonth(parseInt(e.target.value))}>
              {months.filter(m => m >= startMonth).map(m => <option key={m} value={m}>{m} 月</option>)}
            </select>
          </div>
        </div>

        {/* 統計卡片 */}
        <div className="review-stats">
          {[
            { label: '已實現損益', value: totalRealized, color: pnlColor(totalRealized) },
            { label: '股利收入', value: totalDividend, color: '#4f8ef7' },
            { label: '合計收益', value: totalPnL, color: pnlColor(totalPnL) },
          ].map(({ label, value, color }) => (
            <div key={label} style={card}>
              <div style={{ fontSize: '12px', color: '#8892a4', marginBottom: '8px' }}>{label}</div>
              <div style={{ fontSize: 'clamp(18px, 2vw, 22px)', fontWeight: 700, color }}>
                {value >= 0 ? '+' : ''}{value.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}
              </div>
            </div>
          ))}
        </div>

        {/* 長條圖 */}
        <div style={{ ...card, marginBottom: '20px' }}>
          <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '20px' }}>
            {selectedYear} 年 {startMonth}–{endMonth} 月 損益分佈
          </div>
          {loading ? (
            <div style={{ textAlign: 'center', padding: '48px', color: '#8892a4' }}>載入中...</div>
          ) : (
            <ResponsiveContainer width="100%" height={300}>
              <BarChart data={filtered} margin={{ top: 5, right: 20, left: 10, bottom: 5 }}>
                <CartesianGrid strokeDasharray="3 3" stroke="#2e3349" />
                <XAxis dataKey="month" tick={{ fill: '#8892a4', fontSize: 12 }} />
                <YAxis tick={{ fill: '#8892a4', fontSize: 12 }} tickFormatter={v => `${(v / 1000).toFixed(0)}K`} width={50} />
                <Tooltip content={<CustomTooltip />} />
                <Bar dataKey="realized" name="已實現損益" radius={[4, 4, 0, 0]}>
                  {filtered.map((entry, i) => <Cell key={i} fill={entry.realized >= 0 ? '#00c896' : '#ff5757'} />)}
                </Bar>
                <Bar dataKey="dividend" name="股利收入" fill="#4f8ef7" radius={[4, 4, 0, 0]} />
              </BarChart>
            </ResponsiveContainer>
          )}
        </div>

        {/* 月份明細表 */}
        <div style={{ ...card, marginBottom: '40px' }}>
          <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '16px' }}>月份明細</div>
          <div style={{ overflowX: 'auto' }}>
            <table style={{ width: '100%', borderCollapse: 'collapse', fontSize: '14px' }}>
              <thead>
                <tr style={{ borderBottom: '1px solid #2e3349' }}>
                  {['月份', '已實現損益', '股利收入', '合計'].map(h => (
                    <th key={h} style={{ padding: '10px 12px', textAlign: 'left', color: '#8892a4', fontWeight: 500 }}>{h}</th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {filtered.map(m => (
                  <tr key={m.month} style={{ borderBottom: '1px solid #2e3349' }}>
                    <td style={{ padding: '12px', color: '#e2e8f0', fontWeight: 500 }}>{selectedYear} {m.month}</td>
                    <td style={{ padding: '12px', color: pnlColor(m.realized), fontWeight: 600 }}>
                      {m.realized >= 0 ? '+' : ''}{m.realized.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}
                    </td>
                    <td style={{ padding: '12px', color: '#4f8ef7', fontWeight: 600 }}>
                      +{m.dividend.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}
                    </td>
                    <td style={{ padding: '12px', color: pnlColor(m.total), fontWeight: 700 }}>
                      {m.total >= 0 ? '+' : ''}{m.total.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

        {/* ===== 投資績效報告（新增）===== */}
        <div style={{ marginBottom: '24px' }}>
          <h2 style={{ fontSize: 'clamp(16px, 2vw, 20px)', fontWeight: 700, color: '#e2e8f0', margin: '0 0 4px' }}>投資績效報告</h2>
          <p style={{ fontSize: '13px', color: '#8892a4', margin: 0 }}>基於資產趨勢快照計算，請先在「資產趨勢」頁面記錄每年資產</p>
        </div>

        {Object.keys(assetSnapshots).length < 2 ? (
          <div style={{ ...card, textAlign: 'center', padding: '48px', color: '#8892a4' }}>
            <p style={{ fontSize: '15px' }}>資料不足</p>
            <p style={{ fontSize: '13px', marginTop: '8px' }}>請先在「資產趨勢」頁面至少記錄 2 筆以上的資產快照才能計算績效</p>
          </div>
        ) : (
          <>
            {/* 整體績效卡片 */}
            <div className="perf-stats">
              {[
                {
                  label: '整體 CAGR',
                  value: overallCAGR !== null ? `${overallCAGR >= 0 ? '+' : ''}${overallCAGR.toFixed(2)}%` : '—',
                  color: overallCAGR !== null ? pnlColor(overallCAGR) : '#8892a4',
                  desc: '年化複合報酬率'
                },
                {
                  label: '記錄年數',
                  value: `${(() => {
                    const dates = Object.keys(assetSnapshots).sort()
                    if (dates.length < 2) return 0
                    return ((new Date(dates[dates.length - 1]).getTime() - new Date(dates[0]).getTime()) / (365.25 * 24 * 60 * 60 * 1000)).toFixed(1)
                  })()}`,
                  color: '#4f8ef7',
                  desc: '年'
                },
                {
                  label: '最新總資產',
                  value: `$${(assetSnapshots[Object.keys(assetSnapshots).sort().pop()!] || 0).toLocaleString('zh-TW', { maximumFractionDigits: 0 })}`,
                  color: '#4f8ef7',
                  desc: '最近一筆快照'
                },
                {
                  label: '起始資產',
                  value: `$${(assetSnapshots[Object.keys(assetSnapshots).sort()[0]] || 0).toLocaleString('zh-TW', { maximumFractionDigits: 0 })}`,
                  color: '#8892a4',
                  desc: '第一筆快照'
                },
              ].map(({ label, value, color, desc }) => (
                <div key={label} style={card}>
                  <div style={{ fontSize: '12px', color: '#8892a4', marginBottom: '4px' }}>{label}</div>
                  <div style={{ fontSize: 'clamp(18px, 2vw, 22px)', fontWeight: 700, color, marginBottom: '4px' }}>{value}</div>
                  <div style={{ fontSize: '11px', color: '#8892a4' }}>{desc}</div>
                </div>
              ))}
            </div>

            {/* 自訂區間 CAGR */}
            <div style={{ ...card, marginBottom: '20px' }}>
              <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '16px' }}>自訂區間 CAGR</div>
              <div style={{ display: 'flex', gap: '12px', flexWrap: 'wrap', alignItems: 'flex-end' }}>
                <div>
                  <label style={{ fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }}>起始日期</label>
                  <input type="date" style={inp} value={customStart} onChange={e => setCustomStart(e.target.value)} />
                </div>
                <div>
                  <label style={{ fontSize: '12px', color: '#8892a4', display: 'block', marginBottom: '4px' }}>結束日期</label>
                  <input type="date" style={inp} value={customEnd} onChange={e => setCustomEnd(e.target.value)} />
                </div>
                <button onClick={calcCustomCAGR} disabled={perfLoading}
                  style={{ backgroundColor: '#4f8ef7', border: 'none', borderRadius: '8px', padding: '8px 16px', color: '#fff', fontSize: '14px', cursor: 'pointer' }}>
                  計算
                </button>
                {customCAGR !== null && (
                  <div style={{ backgroundColor: customCAGR >= 0 ? 'rgba(0,200,150,0.1)' : 'rgba(255,87,87,0.1)', border: `1px solid ${customCAGR >= 0 ? 'rgba(0,200,150,0.3)' : 'rgba(255,87,87,0.3)'}`, borderRadius: '8px', padding: '8px 20px', display: 'flex', alignItems: 'center', gap: '8px' }}>
                    <span style={{ fontSize: '12px', color: '#8892a4' }}>區間 CAGR</span>
                    <span style={{ fontSize: '22px', fontWeight: 700, color: pnlColor(customCAGR) }}>
                      {customCAGR >= 0 ? '+' : ''}{customCAGR.toFixed(2)}%
                    </span>
                  </div>
                )}
              </div>
            </div>

            {/* 年度報酬率長條圖 */}
            {perfChartData.length >= 2 && (
              <div style={{ ...card, marginBottom: '20px' }}>
                <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '20px' }}>年度報酬率趨勢</div>
                <ResponsiveContainer width="100%" height={280}>
                  <BarChart data={perfChartData} margin={{ top: 5, right: 20, left: 10, bottom: 5 }}>
                    <CartesianGrid strokeDasharray="3 3" stroke="#2e3349" />
                    <XAxis dataKey="year" tick={{ fill: '#8892a4', fontSize: 12 }} />
                    <YAxis tick={{ fill: '#8892a4', fontSize: 12 }} tickFormatter={v => `${v}%`} width={50} />
                    <Tooltip content={<CustomTooltip />} />
                    <Bar dataKey="年度報酬率%" radius={[4, 4, 0, 0]}>
                      {perfChartData.map((entry, i) => (
                        <Cell key={i} fill={entry['年度報酬率%'] >= 0 ? '#00c896' : '#ff5757'} />
                      ))}
                    </Bar>
                  </BarChart>
                </ResponsiveContainer>
              </div>
            )}

            {/* 年度績效明細表 */}
            <div style={card}>
              <div style={{ fontSize: '14px', fontWeight: 600, color: '#e2e8f0', marginBottom: '16px' }}>年度績效明細</div>
              <div style={{ overflowX: 'auto' }}>
                <table className="perf-table">
                  <thead>
                    <tr>
                      {['年度', '年初資產', '年末資產', '年度報酬率', 'CAGR'].map(h => (
                        <th key={h}>{h}</th>
                      ))}
                    </tr>
                  </thead>
                  <tbody>
                    {yearlyPerf.map(p => (
                      <tr key={p.year}>
                        <td style={{ color: '#4f8ef7', fontWeight: 600 }}>{p.year}</td>
                        <td>${p.startValue.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}</td>
                        <td>${p.endValue.toLocaleString('zh-TW', { maximumFractionDigits: 0 })}</td>
                        <td style={{ color: pnlColor(p.totalReturn), fontWeight: 600 }}>
                          {p.totalReturn >= 0 ? '+' : ''}{p.totalReturn.toFixed(2)}%
                        </td>
                        <td style={{ color: pnlColor(p.cagr), fontWeight: 600 }}>
                          {p.cagr >= 0 ? '+' : ''}{p.cagr.toFixed(2)}%
                        </td>
                      </tr>
                    ))}
                    {yearlyPerf.length > 0 && (
                      <tr style={{ borderTop: '2px solid #2e3349', backgroundColor: 'rgba(79,142,247,0.05)' }}>
                        <td style={{ color: '#e2e8f0', fontWeight: 700 }}>整體</td>
                        <td>${(assetSnapshots[Object.keys(assetSnapshots).sort()[0]] || 0).toLocaleString('zh-TW', { maximumFractionDigits: 0 })}</td>
                        <td>${(assetSnapshots[Object.keys(assetSnapshots).sort().pop()!] || 0).toLocaleString('zh-TW', { maximumFractionDigits: 0 })}</td>
                        <td style={{ color: pnlColor((() => {
                          const dates = Object.keys(assetSnapshots).sort()
                          if (dates.length < 2) return 0
                          return ((assetSnapshots[dates[dates.length - 1]] - assetSnapshots[dates[0]]) / assetSnapshots[dates[0]]) * 100
                        })()), fontWeight: 600 }}>
                          {(() => {
                            const dates = Object.keys(assetSnapshots).sort()
                            if (dates.length < 2) return '—'
                            const r = ((assetSnapshots[dates[dates.length - 1]] - assetSnapshots[dates[0]]) / assetSnapshots[dates[0]]) * 100
                            return `${r >= 0 ? '+' : ''}${r.toFixed(2)}%`
                          })()}
                        </td>
                        <td style={{ color: overallCAGR !== null ? pnlColor(overallCAGR) : '#8892a4', fontWeight: 700 }}>
                          {overallCAGR !== null ? `${overallCAGR >= 0 ? '+' : ''}${overallCAGR.toFixed(2)}%` : '—'}
                        </td>
                      </tr>
                    )}
                  </tbody>
                </table>
              </div>
            </div>
          </>
        )}
      </div>
    </>
  )
}
